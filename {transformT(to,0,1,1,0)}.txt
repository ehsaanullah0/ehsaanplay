      {transform:T(to,0,1,1,0)}
    ],{duration:1300,easing:'ease-in-out'});
  }
  const COLORS=['#0a0a0a','#f5f5f5','#a01e29','#7a7a7a','#d9d9d9','#6e0c18'];
  function confetti(){
    for(let i=0;i<34;i++){
      const el=document.createElement('span'); el.className='confetti';
      const size=7+Math.random()*9, shape=Math.random();
      el.style.cssText=`width:${size}px;height:${shape>.66?size*.4:size}px;background:${COLORS[i%COLORS.length]};border-radius:${shape<.4?'50%':shape>.66?'3px':'2px'}`;
      cta.append(el);
      const ang=Math.random()*Math.PI*2, d=170+Math.random()*330;
      const dx=Math.cos(ang)*d*1.5, dy=Math.sin(ang)*d*.8-120, rot=(Math.random()-.5)*900;
      const an=el.animate([
        {transform:'translate(-50%,-50%) scale(.2) rotate(0deg)',opacity:1},
        {transform:`translate(calc(-50% + ${dx*.7}px),calc(-50% + ${dy}px)) scale(1) rotate(${rot*.6}deg)`,opacity:1,offset:.45},
        {transform:`translate(calc(-50% + ${dx}px),calc(-50% + ${dy+260}px)) scale(.8) rotate(${rot}deg)`,opacity:0}
      ],{duration:1000+Math.random()*700,easing:'cubic-bezier(.2,.7,.4,1)',delay:Math.random()*120});
      an.onfinish=()=>el.remove();
    }
  }
  function popLetters(face){
    face.querySelectorAll('h2 .cl').forEach((l,i)=>{
      const s=(i%2?1:-1);
      l.animate([
        {transform:`translateY(46px) scale(.4) rotate(${s*16}deg)`,opacity:0},
        {transform:`translateY(-14px) scale(1.18) rotate(${-s*5}deg)`,opacity:1,offset:.6},
        {transform:'none',opacity:1}
      ],{duration:650,delay:620+i*38,easing:'cubic-bezier(.3,1.4,.5,1)',fill:'backwards'});
    });
  }
  cta.addEventListener('click',e=>{
    const btn=e.target.closest('.swap'); if(!btn||busy) return;
    const f=!cta.classList.contains('flipped');
    const apply=()=>{
      cta.classList.toggle('flipped',f);
      a.toggleAttribute('inert',f); a.setAttribute('aria-hidden',f?'true':'false');
      b.toggleAttribute('inert',!f); b.setAttribute('aria-hidden',f?'false':'true');
    };
    if(!canAnimate){apply();return}
    busy=true; apply();
    setTimeout(()=>{busy=false},1600);
    btn.animate([{transform:'scale(1)'},{transform:'scale(.82) rotate(-6deg)',offset:.3},{transform:'scale(1.12) rotate(4deg)',offset:.65},{transform:'scale(1)'}],{duration:450,easing:'ease-out'});
    const dir=f?1:-1, an=flipAnim(f?0:180,f?180:0,dir);
    setTimeout(confetti,520);
    popLetters(f?b:a);
    an.onfinish=()=>{busy=false};
    an.oncancel=()=>{busy=false};
  });
